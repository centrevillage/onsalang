  f0: F32 at block,
) -> F32 at sample
  pub struct phase_step.State  // Affine; size 16, bulk 0, align 4
    _sample_rate: 4 bytes, align 4, fast  // offset 0
    phase.prev_0: 4 bytes, align 4, fast  // offset 4
    phase._hoist_0: 4 bytes, align 4, fast  // offset 8
    _poisoned: 1 bytes, align 1, fast  // offset 12
    _initialized: 1 bytes, align 1, fast  // offset 13
  pub struct phase_step.Params  // Copy; size 4, align 4
    f0: F32  // offset 0
  pub struct phase_step.Rendered  // Affine
    out: Buf[F32]
  pub const phase_step.SIZE: U32 = 16
  pub const phase_step.BULK_SIZE: U32 = 0
  pub fn phase_step.init(sample_rate: F32) -> phase_step.State
  pub rt fn phase_step.reset(inout s: phase_step.State)
  pub rt fn phase_step.process(inout s: phase_step.State, params: phase_step.Params, inout output: Span[F32])
  pub fn phase_step.render(params: phase_step.Params, frames: U32, sample_rate: F32) -> phase_step.Rendered uses {Alloc}
