package voice

module voice

pub flow resonator(
  x: Sig[F32],
  fc: Ctl[F32],
  bw: Ctl[F32],
) -> Sig[F32]
  pub struct resonator.State  // Affine; size: computed by flow lowering (T3-6)
  pub struct resonator.Config  // Copy; size 0, align 1
  pub struct resonator.Params  // Copy; size 8, align 4
    fc: F32  // offset 0
    bw: F32  // offset 4
  pub struct resonator.Out  // Affine
    out: Buf[F32]
  pub const resonator.SIZE: U32
  pub const resonator.BULK_SIZE: U32
  pub fn resonator.init(cfg: resonator.Config, sample_rate: F32) -> resonator.State
  pub rt fn resonator.reset(inout s: resonator.State)
  pub rt fn resonator.process(inout s: resonator.State, params: resonator.Params, x: Span[F32], inout out: Span[F32])
  pub rt fn resonator.process_inplace(inout s: resonator.State, params: resonator.Params, inout out: Span[F32])
  pub fn resonator.render(cfg: resonator.Config, params: resonator.Params, x: Span[F32], sample_rate: F32) -> resonator.Out uses {Alloc}

pub flow voice(
  @param(min: 20.0, max: 2000.0, default: 110.0, unit: "Hz", scale: "log")
  f0: Ctl[F32],
  @param(min: 200.0, max: 1200.0, default: 700.0, unit: "Hz")
  vowel_f1: Ctl[F32],
  @param(min: 600.0, max: 3000.0, default: 1220.0, unit: "Hz")
  vowel_f2: Ctl[F32],
  @param(min: 0.0, max: 1.0, default: 0.5)
  gain: Ctl[F32],
) -> Sig[F32]
  pub struct voice.State  // Affine; size: computed by flow lowering (T3-6)
  pub struct voice.Config  // Copy; size 0, align 1
  pub struct voice.Params  // Copy; size 16, align 4
    f0: F32  // offset 0
    vowel_f1: F32  // offset 4
    vowel_f2: F32  // offset 8
    gain: F32  // offset 12
  pub struct voice.Out  // Affine
    out: Buf[F32]
  pub const voice.SIZE: U32
  pub const voice.BULK_SIZE: U32
  pub fn voice.init(cfg: voice.Config, sample_rate: F32) -> voice.State
  pub rt fn voice.reset(inout s: voice.State)
  pub rt fn voice.process(inout s: voice.State, params: voice.Params, inout out: Span[F32])
  pub fn voice.render(cfg: voice.Config, params: voice.Params, frames: U32, sample_rate: F32) -> voice.Out uses {Alloc}
  pub fn voice.params_default() -> voice.Params

pub struct Poly  // Affine
  voices: [voice.State; 8]

pub rt fn midi_to_hz(note: U8) -> F32
